  ---
  Proposition : Système de debug par bitmask

  Principe

  Un uint32_t global gDebugFlags dont chaque bit correspond à une catégorie. Les macros DBG(cat, ...) ne s'exécutent que si le bit est actif. Combinaison cumulative avec |.

  Catégories proposées (basées sur le code)

  +-------------------------------------------------------------------+
  ¦    Constante    ¦                     Contenu                     ¦
  +-----------------+-------------------------------------------------¦
  ¦ DBG_MQTT        ¦ Connexions, souscriptions, routage des messages ¦
  +-----------------+-------------------------------------------------¦
  ¦ DBG_CAPTEURS    ¦ DS18B20, DHT20, TCS34725, thermomètres radio    ¦
  +-----------------+-------------------------------------------------¦
  ¦ DBG_ACTIONNEURS ¦ RC devices, volets (TOR), télécommandes RF      ¦
  +-----------------+-------------------------------------------------¦
  ¦ DBG_CHAUDIERE   ¦ Logique chaudière, keep-alive, transitions      ¦
  +-----------------+-------------------------------------------------¦
  ¦ DBG_CONFIG      ¦ Chargement config, méthodes print(), init       ¦
  +-----------------+-------------------------------------------------¦
  ¦ DBG_RESEAU      ¦ WiFi, WebServer                                 ¦
  +-----------------+-------------------------------------------------¦
  ¦ DBG_ECRAN       ¦ Affichage CYD, zones                            ¦
  +-------------------------------------------------------------------+

  Utilisation dans main.cpp

  // Activer cumulativement ce qu'on veut voir :
  gDebugFlags = DBG_MQTT | DBG_CAPTEURS;  // seulement MQTT et capteurs
  gDebugFlags = DBG_ALL;                  // tout voir
  gDebugFlags = DBG_NONE;                 // silence total

  Ce qui serait ajouté dans global.h

  #include <Arduino.h>

  // Catégories de debug (bitmask)
  constexpr uint32_t DBG_NONE        = 0;
  constexpr uint32_t DBG_MQTT        = 1 << 0;
  constexpr uint32_t DBG_CAPTEURS    = 1 << 1;
  constexpr uint32_t DBG_ACTIONNEURS = 1 << 2;
  constexpr uint32_t DBG_CHAUDIERE   = 1 << 3;
  constexpr uint32_t DBG_CONFIG      = 1 << 4;
  constexpr uint32_t DBG_RESEAU      = 1 << 5;
  constexpr uint32_t DBG_ECRAN       = 1 << 6;
  constexpr uint32_t DBG_ALL         = 0xFFFFFFFF;

  extern uint32_t gDebugFlags;

  #define DBG(cat, ...)  do { if (gDebugFlags & (cat)) Serial.printf(__VA_ARGS__); } while(0)
  #define DBGLN(cat, msg) do { if (gDebugFlags & (cat)) Serial.println(msg); } while(0)
